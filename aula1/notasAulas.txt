<input type="text">
	Exibe um campo de entrada de texto de linha única
<input type="radio">
	Exibe um botao de opçao (para selecionar uma das muitas opções)
<input type="checkbox">
	Exibe uma caixa de selecao (para selecionar zero ou mais de muitas opcoes)
<select></select>
	Cria uma lista suspensa.
<input type="buttor">
	Exibe um botão clicável
<textarea>
	Insere uma area de texto para o usuario digitar textos.
</textarea>
